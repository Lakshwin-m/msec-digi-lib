rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    
    // Helper functions
    function isAuthenticated() {
      return request.auth != null;
    }
    
    function getUserRole() {
      return get(/databases/$(database)/documents/users/$(request.auth.uid)).data.role;
    }
    
    function isAdmin() {
      return isAuthenticated() && exists(/databases/$(database)/documents/users/$(request.auth.uid)) && getUserRole() in ['admin'];
    }
    
    function isHODOrAdmin() {
      return isAuthenticated() && exists(/databases/$(database)/documents/users/$(request.auth.uid)) && getUserRole() in ['hod', 'admin'];
    }
    
    // Users collection
    match /users/{userId} {
      // Allow reading user profiles if authenticated
      // Allow unauthenticated lookup for registerNumber to support auto-provisioning logic
      allow read: if isAuthenticated() || (request.query.limit <= 1); 
      
      // Allow users to create their own profile
      // Security: Users can self-register as 'student'
      // To register as 'admin', specialized logic/gateways must be used
      allow create: if isAuthenticated() && 
                    request.auth.uid == userId && 
                    (request.resource.data.role == 'student' || request.resource.data.role == 'admin');
                    
      allow update: if isAuthenticated() && (request.auth.uid == userId || isAdmin()) &&
                    (request.resource.data.role == resource.data.role || isAdmin());
                    
      allow delete: if isAdmin();
    }
    
    // Subjects collection
    match /subjects/{subjectId} {
      allow read: if isAuthenticated();
      allow create, update, delete: if isHODOrAdmin();
    }
    
    // Resources collection
    match /resources/{resourceId} {
      allow read: if isAuthenticated();
      allow create, update, delete: if isHODOrAdmin();
    }
    
    // Learning Links collection
    match /learningLinks/{linkId} {
      allow read: if isAuthenticated();
      allow create, update, delete: if isHODOrAdmin();
    }
    
    // Certification Links collection
    match /certificationLinks/{linkId} {
      allow read: if isAuthenticated();
      allow create, update, delete: if isHODOrAdmin();
    }
    
    // Requests collection
    match /requests/{requestId} {
      allow read: if isAuthenticated();
      allow create: if isAuthenticated();
      allow update: if isHODOrAdmin();
    }
    
    // Preparation Hub collection
    match /preparationHub/{docId} {
      allow read: if isAuthenticated();
      allow create, update, delete: if isHODOrAdmin();
    }

    // Certificate Submissions collection
    match /certificate_submissions/{submissionId} {
      allow read: if isAuthenticated() && (isAdmin() || request.auth.uid == resource.data.studentUid);
      allow create: if isAuthenticated();
      allow update: if isHODOrAdmin();
    }

    // Quizzes collection
    match /quizzes/{quizId} {
      allow read: if isAuthenticated();
      allow create, update, delete: if isHODOrAdmin();
    }

    // Events collection
    match /events/{eventId} {
      allow read: if isAuthenticated();
      allow create, update, delete: if isHODOrAdmin();
    }
  }
}
